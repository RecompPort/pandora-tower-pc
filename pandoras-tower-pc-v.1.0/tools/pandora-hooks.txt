# Funciones del juego que tools/pandora.cpp puede reemplazar (además de las
# del SDK que lista el runtime de wiikit). Cada nombre tiene que estar en
# tools/names-manual.tsv; setup.sh pasa este archivo al recompilador.
pt_SystemMainLoop      # el bucle principal: ritmo de dibujo (60 fps)
pt_StrapTask           # la pantalla de la correa: se puede saltar
MOOpen                 # una película empieza: buscar sus bandas
MOClose                # la película termina
pt_ScreenFrameIn       # las bandas negras de las escenas: se pueden quitar
pt_NowLoadingStart     # la pantalla de carga empieza: sus bandas
pt_NowLoadingEnd       # la pantalla de carga termina
pt_InputShakeUpdate    # las sacudidas que el juego cuenta (para el registro)
